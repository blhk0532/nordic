############################################
# CLEAN LARAVEL OCTANE PRODUCTION STACK
# Nginx + FrankenPHP + Redis + Horizon
############################################

set -e

PROJECT_PATH="/home/ubuntu/nordic"
DOMAIN="nordicdigitalthailand.com"

echo "1. Stop and disable Apache + PHP-FPM..."
sudo systemctl stop apache2 || true
sudo systemctl disable apache2 || true
sudo systemctl stop php8.3-fpm || true
sudo systemctl stop php8.4-fpm || true
sudo systemctl disable php8.3-fpm || true
sudo systemctl disable php8.4-fpm || true

echo "2. Install Nginx + Redis + Supervisor..."
sudo apt update
sudo apt install -y nginx redis-server supervisor

echo "3. Enable Redis..."
sudo systemctl enable redis
sudo systemctl start redis

echo "4. Fix project permissions..."
sudo chown -R ubuntu:www-data $PROJECT_PATH
sudo find $PROJECT_PATH -type f -exec chmod 664 {} \;
sudo find $PROJECT_PATH -type d -exec chmod 775 {} \;
sudo chmod -R 775 $PROJECT_PATH/storage
sudo chmod -R 775 $PROJECT_PATH/bootstrap/cache
sudo usermod -aG www-data ubuntu

echo "5. Configure Nginx..."
sudo tee /etc/nginx/sites-available/nordic <<EOF
server {
    listen 80;
    server_name $DOMAIN;

    root $PROJECT_PATH/public;

    location /build/ {
        try_files \$uri \$uri/ =404;
    }

    location /storage/ {
        try_files \$uri \$uri/ =404;
    }

    location / {
        proxy_pass http://127.0.0.1:8000;
        proxy_http_version 1.1;
        proxy_set_header Host \$host;
        proxy_set_header X-Forwarded-Proto \$scheme;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
    }
}
EOF

sudo ln -sf /etc/nginx/sites-available/nordic /etc/nginx/sites-enabled/nordic
sudo nginx -t
sudo systemctl restart nginx
sudo systemctl enable nginx

echo "6. Setup Supervisor for FrankenPHP (Octane)..."
sudo tee /etc/supervisor/conf.d/octane.conf <<EOF
[program:octane]
command=/usr/local/bin/frankenphp run --config=$PROJECT_PATH/vendor/laravel/octane/src/Commands/stubs/Caddyfile
directory=$PROJECT_PATH
user=ubuntu
autostart=true
autorestart=true
redirect_stderr=true
stdout_logfile=/var/log/octane.log
EOF

echo "7. Setup Supervisor for Horizon..."
sudo tee /etc/supervisor/conf.d/horizon.conf <<EOF
[program:horizon]
command=/usr/bin/php $PROJECT_PATH/artisan horizon
directory=$PROJECT_PATH
user=ubuntu
autostart=true
autorestart=true
redirect_stderr=true
stdout_logfile=/var/log/horizon.log
EOF

echo "8. Reload Supervisor..."
sudo supervisorctl reread
sudo supervisorctl update
sudo supervisorctl restart all

echo "9. Clear Laravel caches..."
cd $PROJECT_PATH
php artisan optimize:clear

echo "10. Done."
echo "Stack is now:"
echo "Nginx -> FrankenPHP (Octane) -> Laravel -> Redis -> Horizon"
